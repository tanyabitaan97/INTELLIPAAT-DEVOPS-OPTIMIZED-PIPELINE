Terraform installation 
To install Terraform on Amazon Linux 2023 EC2, run the following commands:

1. Update the system
sudo dnf update -y
2. Install required packages
sudo dnf install -y yum-utils unzip
3. Add HashiCorp repository
sudo dnf config-manager --add-repo https://rpm.releases.hashicorp.com/AmazonLinux/hashicorp.repo
4. Install Terraform
sudo dnf -y install terraform
5. Verify installation
terraform version

Jenkins



1. Install Jenkins on Bastion

Create jenkins.yml

---
- hosts: bastion
  become: yes

  tasks:

    - name: Install Java 17
      dnf:
        name: java-17-amazon-corretto
        state: present

    - name: Add Jenkins Repository
      get_url:
        url: https://pkg.jenkins.io/redhat-stable/jenkins.repo
        dest: /etc/yum.repos.d/jenkins.repo

    - name: Import Jenkins Key
      rpm_key:
        state: present
        key: https://pkg.jenkins.io/redhat-stable/jenkins.io-2023.key

    - name: Install Jenkins
      dnf:
        name: jenkins
        state: present

    - name: Start Jenkins
      service:
        name: jenkins
        state: started
        enabled: yes

8. Create Jenkins Pipeline Job

Repository:

Employee Portal Repository

Pipeline Script:

pipeline {

    agent any

    environment {
        IMAGE_NAME = "employee-portal"
        DOCKER_USER = "YOUR_DOCKERHUB_USER"
    }

    stages {

        stage('Clone Code') {
            steps {
                git 'https://github.com/Kiaraintellipaat/Employee-Portal-with-Automated-DevOps-Delivery-Pipeline.git'
            }
        }

        stage('Build Docker Image') {
            steps {
                sh 'docker build -t $DOCKER_USER/$IMAGE_NAME:$BUILD_NUMBER .'
            }
        }

        stage('Push Docker Image') {
            steps {
                withCredentials([
                    usernamePassword(
                        credentialsId: 'dockerhub',
                        usernameVariable: 'DOCKER_USERNAME',
                        passwordVariable: 'DOCKER_PASSWORD'
                    )
                ]) {

                    sh '''
                    docker login -u $DOCKER_USERNAME -p $DOCKER_PASSWORD
                    docker push $DOCKER_USER/$IMAGE_NAME:$BUILD_NUMBER
                    '''
                }
            }
        }
    }
}
