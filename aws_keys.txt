cat Dockerfile

    }

        steps { 

            sh ''' 

            docker build -t $IMAGE_NAME:$IMAGE_TAG . 

            ''' 

        }

        steps {

Pipeline Script:

pipeline {

    agent any

    environment {
        IMAGE_NAME = "employee-portal"
        DOCKER_USER = "YOUR_DOCKERHUB_USER"
    }

    stages {

        stage('Clone Code') {
            steps {
                git 'https://github.com/Kiaraintellipaat/Employee-Portal-with-Automated-DevOps-Delivery-Pipeline.git'
            }
        }

        stage('Build Docker Image') {
            steps {
                sh 'docker build -t $DOCKER_USER/$IMAGE_NAME:$BUILD_NUMBER .'
            }
        }

        stage('Push Docker Image') {
            steps {
                withCredentials([
                    usernamePassword(
                        credentialsId: 'dockerhub',
                        usernameVariable: 'DOCKER_USERNAME',
                        passwordVariable: 'DOCKER_PASSWORD'
                    )
                ]) {

                    sh '''
                    docker login -u $DOCKER_USERNAME -p $DOCKER_PASSWORD
                    docker push $DOCKER_USER/$IMAGE_NAME:$BUILD_NUMBER
                    '''
                }
            }
        }
    }
}
